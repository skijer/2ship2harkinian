"""Copy Ganondorf's cloak (OoT En_Ganon_Mant) into this mod: 2Ship's oot.o2r has no overlay assets."""
from pathlib import Path
import struct
import zipfile

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[4]
ARCHIVE = ROOT / 'Shipwright-unbound/x64/Release/oot.o2r'
SOURCE = 'overlays/ovl_En_Ganon_Mant/'
TARGET = 'objects/object_nei_magic_cape/'
NAMES = ('gMantTex', 'gMant1Vtx', 'gMant2Vtx', 'gMantMaterialDL', 'gMantDL')
HASHED_OPCODES = (0x20, 0x31, 0x32, 0x04, 0x33, 0x36, 0x37)
OUT = HERE / 'assets' / TARGET


def crc(path):
    value = (1 << 64) - 1
    for byte in path.encode():
        value ^= byte << 56
        for _ in range(8):
            value = ((value << 1) ^ (0x42F0E1EBA9EA3693 if value >> 63 else 0)) & ((1 << 64) - 1)
    return value


def remap_display_list(raw, remap):
    pos = 72
    while pos < len(raw):
        w0, = struct.unpack_from('<I', raw, pos)
        op = w0 >> 24
        if op in HASHED_OPCODES:
            high, low = struct.unpack_from('<II', raw, pos + 8)
            target = remap.get((high << 32) | low)
            if target is not None:
                struct.pack_into('<II', raw, pos + 8, target >> 32, target & 0xFFFFFFFF)
            pos += 16
        else:
            pos += 8
        if op == 0xDF:
            break
    return raw


OUT.mkdir(parents=True, exist_ok=True)
remap = {crc(SOURCE + name): crc(TARGET + name) for name in NAMES}
with zipfile.ZipFile(ARCHIVE) as archive:
    for name in NAMES:
        raw = bytearray(archive.read(SOURCE + name))
        if name.endswith('DL'):
            raw = remap_display_list(raw, remap)
        (OUT / name).write_bytes(raw)
print('Magic cape: copied', ', '.join(NAMES))
